POSITIVE OR NEGATIVE CHECKER

namespace IfSampleDEMO
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnCheck_Click(object sender, EventArgs e)
        {
            int number = Convert.ToInt32(txtNumber.Text);

            if (number > 0)
            {
                lblResult.Text = "Result: The number is positive.";
            }
            else if (number < 0)
            {
                lblResult.Text = "Result: The number is negative.";
            }
            else
            {
                lblResult.Text = "Result: The number is zero.";
            }
        }

        {

        }
    }
