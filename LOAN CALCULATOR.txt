LOAN CALCULATOR

Console.Write("Enter Loan Amount: ");
double loan = Convert.ToDouble(Console.ReadLine());

Console.Write("Enter Annual Interest Rate: ");
double interestRate = Convert.ToDouble(Console.ReadLine());

Console.Write("Years to Pay Off: ");
int years = Convert.ToInt32(Console.ReadLine());

double totalInterest = loan * interestRate * years;
double totalAmount = loan + totalInterest;
double monthlyPayment = totalAmount / (years * 12);

bool isValid = loan >= 100000 && interestRate > 20000 && monthlyPayment >= 5000;

Console.WriteLine("\n======================================");
Console.WriteLine("\nLoan Amount: " + loan);
Console.WriteLine("Annual Interest Rate: " + interestRate); 
Console.WriteLine("Monthly Payment: " + monthlyPayment);
Console.WriteLine("\n======================================");
Console.WriteLine("\nLoan Amount is valid: " + isValid);
Console.WriteLine("Annual Interest Rate is valid: " + (interestRate > 20000));
Console.WriteLine("Monthly Payment is valid: " + (monthlyPayment >= 5000));
Console.WriteLine("\n======================================");
